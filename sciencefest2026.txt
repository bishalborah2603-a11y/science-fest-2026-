// =================================
// CONFIGURATION

const ADMIN_PIN = "SF2026";

const STORAGE_KEY = "scienceFest2026Registrations";

const OPEN_DATE = new Date("2026-11-10T00:00:00+05:30");
const CLOSE_DATE = new Date("2026-11-27T00:00:00+05:30");
const FEST_START = new Date("2026-11-30T00:00:00+05:30");
const FEST_END = new Date("2026-12-06T00:00:00+05:30");

const $ = id => document.getElementById(id);

// =================================
// MOBILE NAVIGATION
// =================================

$("menuBtn").addEventListener("click", () => {
 $("nav").classList.toggle("active");
});

document.querySelectorAll("nav a").forEach(link => {
 link.addEventListener("click", () => {
  $("nav").classList.remove("active");
 });
});

// =================================
// COUNTDOWN
// =================================

function updateCountdown() {
 const now = new Date();
 let target;
 let title;

 if (now < OPEN_DATE) {
  target = OPEN_DATE;
  title = "REGISTRATION OPENS IN";
 } else if (now < CLOSE_DATE) {
  target = CLOSE_DATE;
  title = "REGISTRATION CLOSES IN";
 } else if (now < FEST_START) {
  target = FEST_START;
  title = "SCIENCE FEST BEGINS IN";
 } else if (now < FEST_END) {
  target = FEST_END;
  title = "FEST CONCLUDES IN";
 } else {
  $("countTitle").textContent = "THANK YOU FOR JOINING SCIENCE FEST 2026!";
  $("countdown").style.display = "none";
  return;
 }

 $("countTitle").textContent = title;

 const distance = Math.max(0, target - now);

 $("days").textContent = String(Math.floor(distance / 86400000)).padStart(2,"0");
 $("hours").textContent = String(Math.floor(distance % 86400000 / 3600000)).padStart(2,"0");
 $("minutes").textContent = String(Math.floor(distance % 3600000 / 60000)).padStart(2,"0");
 $("seconds").textContent = String(Math.floor(distance % 60000 / 1000)).padStart(2,"0");
}

updateCountdown();
setInterval(updateCountdown,1000);

// =================================
// REGISTRATION STORAGE
// =================================

function getRegistrations() {
 try {
  const data = JSON.parse(localStorage.getItem(STORAGE_KEY) || "[]");
  return Array.isArray(data) ? data : [];
 } catch {
  return [];
 }
}

function saveRegistrations(data) {
 localStorage.setItem(STORAGE_KEY,JSON.stringify(data));
}

function updateRegistrationStatus() {
 const now = new Date();

 if (now < OPEN_DATE) {
  $("submitBtn").disabled = true;
  $("submitBtn").textContent = "REGISTRATION OPENS 10 NOVEMBER";
  $("message").textContent = "Registration has not opened yet.";
 } else if (now >= CLOSE_DATE) {
  $("submitBtn").disabled = true;
  $("submitBtn").textContent = "REGISTRATION CLOSED";
  $("message").textContent = "The registration deadline has passed.";
 } else {
  $("submitBtn").disabled = false;
  $("submitBtn").textContent = "SUBMIT REGISTRATION →";
  $("message").textContent = "";
 }
}

updateRegistrationStatus();
setInterval(updateRegistrationStatus,30000);

// =================================
// SUBMIT REGISTRATION
// =================================

$("registrationForm").addEventListener("submit", event => {
 event.preventDefault();

 const now = new Date();

 if (now < OPEN_DATE || now >= CLOSE_DATE) {
  $("message").textContent = "Registration is currently closed.";
  updateRegistrationStatus();
  return;
 }

 const registration = {
  id: "SF2026-" + Date.now().toString(36).toUpperCase(),
  name: $("name").value.trim(),
  email: $("email").value.trim(),
  phone: $("phone").value.trim(),
  school: $("school").value.trim(),
  category: $("category").value,
  idea: $("idea").value.trim(),
  submittedAt: now.toISOString()
 };

 if (!registration.name || !registration.email ||
     !registration.phone || !registration.school ||
     !registration.category) {
  $("message").textContent = "Please fill in all required fields.";
  return;
 }

 try {
  const registrations = getRegistrations();

  registrations.push(registration);
  saveRegistrations(registrations);

  $("message").textContent =
   "Registration saved! Your demo ID: " + registration.id;

  $("registrationForm").reset();

 } catch(error) {
  $("message").textContent = "Unable to save registration.";
 }
});

// RENDER ADMIN DASHBOARD
// =================================

function makeCell(value) {
 const td = document.createElement("td");
 td.textContent = value || "-";
 return td;
}

function renderDashboard() {
 const all = getRegistrations();
 const tech = all.filter(x => x.category === "AI & Digital Innovation").length;
 const robotics = all.filter(x => x.category === "Robotics & Automation").length;

 $("totalCount").textContent = all.length;
 $("techCount").textContent = tech;
 $("roboticsCount").textContent = robotics;
 $("otherCount").textContent = all.length - tech - robotics;

 const search = $("searchInput").value.toLowerCase().trim();
 const filter = $("filterCategory").value;

 const filtered = all.filter(item => {
  const text = [
   item.name,item.email,item.school,item.phone,item.id
  ].join(" ").toLowerCase();

  return text.includes(search) &&
   (!filter || item.category === filter);
 });

 const tbody = $("registrationRows");
 tbody.replaceChildren();

 $("emptyMessage").style.display = filtered.length ? "none" : "block";

 filtered.forEach(item => {
  const tr = document.createElement("tr");

  [
   item.id,item.name,item.email,item.phone,
   item.school,item.category,item.idea
  ].forEach(value => tr.appendChild(makeCell(value)));

  const actionCell = document.createElement("td");
  const deleteButton = document.createElement("button");

  deleteButton.className = "delete-btn";
  deleteButton.textContent = "DELETE";

  deleteButton.addEventListener("click", () => {
   if (!confirm("Delete this registration?")) return;

   saveRegistrations(
    getRegistrations().filter(x => x.id !== item.id)
   );

   renderDashboard();
  });

  actionCell.appendChild(deleteButton);
  tr.appendChild(actionCell);
  tbody.appendChild(tr);
 });
}

$("searchInput").addEventListener("input",renderDashboard);
$("filterCategory").addEventListener("change",renderDashboard);

$("exportBtn").addEventListener("click", () => {
 const registrations = getRegistrations();

 if (!registrations.length) {
  alert("There are no registrations to export.");
  return;
 }

 const columns = [
  "id","name","email","phone","school",
  "category","idea","submittedAt"
 ];

 function escapeCSV(value) {
  let text = String(value ?? "");

  if (/^[=+\-@\t\r]/.test(text)) {
   text = "'" + text;
  }

  return '"' + text.replace(/"/g,'""') + '"';
 }

 const csv = [
  columns.map(escapeCSV).join(","),
  ...registrations.map(row =>
   columns.map(col => escapeCSV(row[col])).join(",")
  )
 ].join("\r\n");

 const blob = new Blob(["\uFEFF" + csv], {
  type:"text/csv;charset=utf-8;"
 });

 const url = URL.createObjectURL(blob);
 const link = document.createElement("a");

 link.href = url;
 link.download = "ScienceFest2026-Registrations.csv";
 link.click();

 URL.revokeObjectURL(url);
});
